-- simple queries on sales and revenue system --
-- 1. Display all records from the CUSTOMERS table. 
 SELECT * FROM customers;
-- 2. Display all product names and prices from the PRODUCTS table. 
SELECT product_name, price
FROM products;
-- 3. Find products whose price is greater than 1,000.
SELECT *FROM products
WHERE price > 1000;
-- 4. Display customers whose names start with 'A'.
SELECT *FROM customers
WHERE customer_name LIKE 'A%';
-- 5. Find all orders placed on a particular date.
SELECT *FROM orders
WHERE order_date = '2026-10-17';
-- 6. Display products in descending order of price.
SELECT *FROM products
ORDER BY price DESC;
-- 7. Find the total number of customers.
SELECT COUNT(*) AS total_customers
FROM customers;
-- 8. Find the minimum and maximum product prices.
SELECT MIN(price) AS minimum_price,
       MAX(price) AS maximum_price FROM products;
-- 9. Display the total quantity sold from ORDER_DETAILS.
SELECT SUM(quantity) AS total_quantity_sold
FROM order_details;
-- 10. Find the total revenue from all orders using the ORDERS table.
SELECT SUM(price * quantity) AS total_revenue
FROM order_details;
