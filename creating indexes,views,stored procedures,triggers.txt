-- Customer Order Details Using Views --
CREATE VIEW VW_CUSTOMER_ORDERS AS
SELECT
    C.CUSTOMER_ID,
    C.CUSTOMER_NAME,
    O.ORDER_ID,
    O.ORDER_DATE
FROM CUSTOMERS C
INNER JOIN ORDERS O
ON C.CUSTOMER_ID = O.CUSTOMER_ID;
SELECT * FROM VW_CUSTOMER_ORDERS;
-- Product Sales Details using views --
CREATE VIEW VW_PRODUCT_SALES AS
SELECT
    P.PRODUCT_ID,
    P.PRODUCT_NAME,
    P.CATEGORY,
    SUM(OD.QUANTITY) AS TOTAL_QUANTITY,
    SUM(OD.PRICE * OD.QUANTITY) AS TOTAL_REVENUE
FROM PRODUCTS P
INNER JOIN ORDER_DETAILS OD
ON P.PRODUCT_ID = OD.PRODUCT_ID
GROUP BY
    P.PRODUCT_ID,
    P.PRODUCT_NAME,
    P.CATEGORY;
 SELECT * FROM VW_PRODUCT_SALES;
 -- Monthly Revenue using views --
CREATE VIEW VW_MONTHLY_REVENUE AS
SELECT
    YEAR(O.ORDER_DATE) AS ORDER_YEAR,
    MONTH(O.ORDER_DATE) AS ORDER_MONTH,
    SUM(OD.PRICE * OD.QUANTITY) AS TOTAL_REVENUE
FROM ORDERS O
INNER JOIN ORDER_DETAILS OD
ON O.ORDER_ID = OD.ORDER_ID
GROUP BY
    YEAR(O.ORDER_DATE),
    MONTH(O.ORDER_DATE);
SELECT * FROM VW_MONTHLY_REVENUE;

-- Get Customer Orders Using Parameters in stored procedures --
DELIMITER //
CREATE PROCEDURE SP_CUSTOMER_ORDERS(
    IN P_CUSTOMER_ID INT)
BEGIN
    SELECT
        C.CUSTOMER_NAME,
        O.ORDER_ID,
        O.ORDER_DATE
    FROM CUSTOMERS C
    INNER JOIN ORDERS O
    ON C.CUSTOMER_ID = O.CUSTOMER_ID
    WHERE C.CUSTOMER_ID = P_CUSTOMER_ID;
END //
DELIMITER ;
CALL SP_CUSTOMER_ORDERS(103);
-- Product-Wise Revenue using stored procedures --
DELIMITER //
CREATE PROCEDURE SP_PRODUCT_REVENUE(
    IN P_PRODUCT_ID INT)
BEGIN
    SELECT
        P.PRODUCT_NAME,
        SUM(OD.QUANTITY) AS TOTAL_QUANTITY,
        SUM(OD.PRICE * OD.QUANTITY) AS TOTAL_REVENUE
    FROM PRODUCTS P
    INNER JOIN ORDER_DETAILS OD
    ON P.PRODUCT_ID = OD.PRODUCT_ID
    WHERE P.PRODUCT_ID = P_PRODUCT_ID
    GROUP BY P.PRODUCT_ID, P.PRODUCT_NAME;
END //
DELIMITER ;
CALL SP_PRODUCT_REVENUE(205);
-- Monthly Revenue Between Two Dates by using stored procedures --
DELIMITER //
CREATE PROCEDURE SP_REVENUE_BY_DATE(
    IN P_START_DATE DATE,
    IN P_END_DATE DATE)
BEGIN
    SELECT
        SUM(OD.PRICE * OD.QUANTITY) AS TOTAL_REVENUE
    FROM ORDERS O
    INNER JOIN ORDER_DETAILS OD
    ON O.ORDER_ID = OD.ORDER_ID
    WHERE O.ORDER_DATE BETWEEN P_START_DATE
    AND P_END_DATE;
END //
DELIMITER ;
CALL SP_REVENUE_BY_DATE('2026-10-01','2026-10-31');

-- Triggers --
-- BEFORE INSERT Stock Quantity or Price must be greater than Zero--
DELIMITER //
CREATE TRIGGER TRG_BEFORE_ORDER_INSERT
BEFORE INSERT ON ORDER_DETAILS
FOR EACH ROW
BEGIN
    IF NEW.QUANTITY <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Quantity must be greater than zero';
    END IF;
    IF NEW.PRICE <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Price must be greater than zero';
    END IF;
END //
DELIMITER ;
-- Should prevent customers from purchasing more products than available stock --
DELIMITER //
CREATE TRIGGER TRG_CHECK_STOCK
BEFORE INSERT ON ORDER_DETAILS
FOR EACH ROW
BEGIN
    DECLARE AVAILABLE_STOCK INT;
    SELECT STOCK_QUANTITY
    INTO AVAILABLE_STOCK
    FROM PRODUCTS
    WHERE PRODUCT_ID = NEW.PRODUCT_ID
    FOR UPDATE;
    IF AVAILABLE_STOCK IS NULL THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Product does not exist';

    ELSEIF AVAILABLE_STOCK < NEW.QUANTITY THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Insufficient stock available';
    END IF;
END //
delimiter;
